import pandas as pd, numpy as np
from imblearn.over_sampling import SMOTE
p=pd.read_csv('data/interim/train_prepared.csv'); fin=pd.read_csv('data/processed/stroke_clean_final.csv')
X=p.drop(columns=['work_type']); y=p.work_type
mk=lambda: SMOTE(k_neighbors=5,sampling_strategy='auto',random_state=33)
Xf,yf=mk().fit_resample(X.astype(float),y)
lab=pd.Series(yf).value_counts().sort_index()
oh=fin[[c for c in fin if c.startswith('work_type_')]].sum()
print(pd.DataFrame({'label count':lab.values,'one hot sum':oh.values},index=oh.index))
print((fin[oh.index].idxmax(axis=1).str.replace('work_type_','').values==np.asarray(yf)).all())
Xl,yl=mk().fit_resample(X,y)                       # as loaded (int64 0/1 columns)
int_cols=[c for c in X if X[c].dtype.kind=='i']
print((Xl[int_cols].values==np.trunc(Xf[int_cols]).astype(int).values).all())
s=Xf['stroke']; r=s[(s>0.5)&(s<1)].index[0]
print(r, round(s[r],6), Xl.loc[r,'stroke'], int(np.trunc(s[r])), int(s[r]>=0.5))
m=(Xf[int_cols]>=0.5)&(Xf[int_cols]<1)
print(int(m.sum().sum()), int((m&(Xl[int_cols]==0)).sum().sum()))
