Script B (independent rebuild of prepared train; leakage on state):
```python
import pandas as pd, numpy as np, joblib
tr=pd.read_csv('data/interim/train_raw.csv'); te=pd.read_csv('data/interim/test_raw.csv')
st=joblib.load('data/interim/prep_state.joblib')
num=['age','avg_glucose_level','bmi']; txt=['gender','ever_married','Residence_type','smoking_status']
def prep(d, med, mu, sd, cats):
    o=d.drop(columns=['id']).copy()
    o['bmi_missing']=o.bmi.isna().astype(int); o['bmi']=o.bmi.fillna(med)
    oh=[(f"{c}_{v}", (o[c]==v).astype(int)) for c in txt for v in cats[c]]
    wt=o.work_type; o=o.drop(columns=txt+['work_type'])
    for i,c in enumerate(num): o[c]=(o[c]-mu[i])/sd[i]
    for n,s in oh: o[n]=s
    o['work_type']=wt; return o
tr2=tr[tr.gender!='Other']; med=tr2.bmi.median()
f=tr2[num].copy(); f['bmi']=f.bmi.fillna(med); mu=f.mean().values; sd=f.std(ddof=0).values
cats={c:sorted(tr2[c].unique()) for c in txt}
mine=prep(tr2,med,mu,sd,cats).reset_index(drop=True)
saved=pd.read_csv('data/interim/train_prepared.csv')
print(saved.shape, mine.shape, list(saved.columns)==list(mine.columns))
pd.testing.assert_frame_equal(saved, mine, check_dtype=False, atol=1e-9, rtol=0)
print({c: saved[[x for x in saved if x.startswith(c+'_')]].sum(axis=1).value_counts().to_dict() for c in txt})
```
